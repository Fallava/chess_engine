#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
pub struct Square(u8);

impl Square {
    pub fn new(index: u8) -> Option<Square> {
        if index < 64 {
            Some(Square(index))
        } else {
            None
        }
    }

    pub fn index(self) -> usize {
        self.0 as usize
    }

    pub fn file(self) -> u8 {
        self.0 % 8
    }

    pub fn rank(self) -> u8 {
        self.0 / 8
    }

    pub fn to_uci(self) -> String {
        let letter = (b'a' + self.file()) as char;
        let digit = (b'1' + self.rank()) as char;
        format!("{letter}{digit}")
    }

    pub fn from_uci(s: &str) -> Option<Square> {
        // Exactement 2 octets, sinon None. Aucun accès par index : impossible de paniquer.
        let &[file, rank] = s.as_bytes() else {
            return None;
        };

        let file = file.wrapping_sub(b'a');
        let rank = rank.wrapping_sub(b'1');

        if file < 8 && rank < 8 {
            Some(Square(rank * 8 + file))
        } else {
            None
        }
    }
}

pub enum Piece {
    King,
    Queen,
    Pawn,
    Rook,
    Bishop,
    Knight,
}

impl Piece {
    pub const ALL: [Piece; 6] = [
        Piece::Pawn, Piece::Knight, Piece::Bishop,
        Piece::Rook, Piece::Queen, Piece::King,
    ];

    pub fn index(self) -> usize {
        self as usize
    }
    
    pub fn from_char(c: char) -> Option<Self> {
        match c {
            'k' => Some(Piece::King),
            'q' => Some(Piece::Queen),
            'p' => Some(Piece::Pawn),
            'r' => Some(Piece::Rook),
            'b' => Some(Piece::Bishop),
            'n' => Some(Piece::Knight),
            _ => None
        }
    }

    pub fn to_char(self) -> char {
        match self {
            Piece::King     => 'k',
            Piece::Queen    => 'q',
            Piece::Pawn     => 'p',
            Piece::Rook     => 'r',
            Piece::Bishop   => 'b',
            Piece::Knight   => 'n',
        }
    }
}

use std::ops::{
    BitOr, BitOrAssign,
    BitAnd, BitAndAssign,
    BitXor, BitXorAssign,

use super::square::Square;

#[derive(Clone, Copy, PartialEq, Eq, Hash, Debug)]
pub struct Bitboard(u64);

impl Bitboard {
    pub const EMPTY: Bitboard = Bitboard(0);

    pub const FILE_A: Bitboard = Bitboard(0x0101_0101_0101_0101);
    pub const FILE_B: Bitboard = Bitboard(0x0202_0202_0202_0202);
    pub const FILE_C: Bitboard = Bitboard(0x0404_0404_0404_0404);
    pub const FILE_D: Bitboard = Bitboard(0x0808_0808_0808_0808);
    pub const FILE_E: Bitboard = Bitboard(0x1010_1010_1010_1010);
    pub const FILE_F: Bitboard = Bitboard(0x2020_2020_2020_2020);
    pub const FILE_G: Bitboard = Bitboard(0x4040_4040_4040_4040);
    pub const FILE_H: Bitboard = Bitboard(0x8080_8080_8080_8080);

    pub const RANK_1: Bitboard = Bitboard(0x0000_0000_0000_00FF);
    pub const RANK_2: Bitboard = Bitboard(0x0000_0000_0000_FF00);
    pub const RANK_3: Bitboard = Bitboard(0x0000_0000_00FF_0000);
    pub const RANK_4: Bitboard = Bitboard(0x0000_0000_FF00_0000);
    pub const RANK_5: Bitboard = Bitboard(0x0000_00FF_0000_0000);
    pub const RANK_6: Bitboard = Bitboard(0x0000_FF00_0000_0000);
    pub const RANK_7: Bitboard = Bitboard(0x00FF_0000_0000_0000);
    pub const RANK_8: Bitboard = Bitboard(0xFF00_0000_0000_0000);

    pub const fn new(bits: u64) -> Bitboard { Bitboard(bits) }
    pub const fn bits(self) -> u64 { self.0 }

    pub fn from_square(sq: Square) -> Self {
        Bitboard(1 << sq.index())
    }

    pub fn contains(self, sq: Square) -> bool {
        (self.0 >> sq.index()) & 1 == 1
    }

    pub fn insert(&mut self, sq: Square) {
        let mask = Self::from_square(sq);
        self.0 |= mask.0;
    }

    pub fn remove(&mut self, sq: Square) {
        let mask = Self::from_square(sq);
        self.0 &= !mask.0;
    }

    pub fn count(self) -> u32 {
        self.0.count_ones()
    }

    pub fn is_empty(self) -> bool {
        self.0 == 0
    }

    pub fn north(self) -> Bitboard {
        Bitboard(self.0 << 8)
    }

    pub fn south(self) -> Bitboard {
        Bitboard(self.0 >> 8)
    }

    pub fn east(self) -> Bitboard {
        Bitboard(self.0 << 1 & !Self::FILE_A.0)
    }

    pub fn west(self) -> Bitboard {
        Bitboard(self.0 >> 1 & !Self::FILE_H.0)
    }

    pub fn north_east(self) -> Bitboard {
        Bitboard(self.0 << 9 & !Self::FILE_A.0)
    }

    pub fn north_west(self) -> Bitboard {
        Bitboard(self.0 << 7 & !Self::FILE_H.0)
    }

    pub fn south_east(self) -> Bitboard {
        Bitboard(self.0 >> 7 & !Self::FILE_A.0)
    }

    pub fn south_west(self) -> Bitboard {
        Bitboard(self.0 >> 9 & !Self::FILE_H.0)
    }

    pub fn lsb(self) -> Option<Square> {
        let index = self.0.trailing_zeros() as u8;
        Square::new(index)
    }

    pub fn pop_lsb(&mut self) -> Option<Square> {
        let sq = self.lsb()?;
        self.0 &= self.0 - 1;
        Some(sq)
    } 
}

impl Iterator for Bitboard {
    type Item = Square;

    fn next(&mut self) -> Option<Square> {
        self.pop_lsb()
    }
}

impl BitOr for Bitboard {
    type Output = Self;
    fn bitor(self, rhs: Self) -> Self::Output {
        Bitboard(self.0 | rhs.0)
    }
}

impl BitOrAssign for Bitboard {
    fn bitor_assign(&mut self, rhs: Self) {
        self.0 |= rhs.0
    }
}

impl BitAnd for Bitboard {
    type Output = Self;
    fn bitand(self, rhs: Self) -> Self::Output {
        Bitboard(self.0 & rhs.0)
    }
}

impl BitAndAssign for Bitboard {
    fn bitand_assign(&mut self, rhs: Self) {
        self.0 &= rhs.0
    }
}

impl BitXor for Bitboard {
    type Output = Self;
    fn bitxor(self, rhs: Self) -> Self::Output {
        Bitboard(self.0 ^ rhs.0)
    }
}

impl BitXorAssign for Bitboard {
    fn bitxor_assign(&mut self, rhs: Self) {
        self.0 ^= rhs.0
    }
}

impl Not for Bitboard {
    type Output = Self;
    fn not(self) -> Self::Output {
        Bitboard(!self.0)
    }
}

use super::bitboard::Bitboard;
use super::color::Color;
use super::square::Square;

pub fn king_attacks(sq: Square) -> Bitboard {
    let b = Bitboard::from_square(sq);
    b.north() | b.south() | b.east() | b.west() | b.north_east() | b.north_west() | b.south_east() | b.south_west()
}

pub fn knight_attacks(sq: Square) -> Bitboard {
    let b = Bitboard::from_square(sq);
    b.north().north_east() | b.north().north_west() 
    | b.east().north_east() | b.west().north_west()
    | b.south().south_east() | b.south().south_west() 
    | b.east().south_east() | b.west().south_west()
}

pub fn pawn_attacks(color: Color, sq: Square) -> Bitboard {
    let b = Bitboard::from_square(sq);
    match color {
        Color::White => {
            b.north_east() | b.north_west()
        },
        Color::Black => {
            b.south_east() | b.south_west()
        }
    }
}

pub fn rook_attacks(sq: Square, occupied: Bitboard) -> Bitboard {
    let s1 = slide(sq, occupied, Bitboard::north);
    let s2 = slide(sq, occupied, Bitboard::south);
    let s3 = slide(sq, occupied, Bitboard::east);
    let s4 = slide(sq, occupied, Bitboard::west);
    s1 | s2 | s3 | s4
}

pub fn bishop_attacks(sq: Square, occupied: Bitboard) -> Bitboard{
    let s1 = slide(sq, occupied, Bitboard::north_east);
    let s2 = slide(sq, occupied, Bitboard::north_west);
    let s3 = slide(sq, occupied, Bitboard::south_east);
    let s4 = slide(sq, occupied, Bitboard::south_west);
    s1 | s2 | s3 | s4
}

pub fn queen_attacks(sq: Square, occupied: Bitboard) -> Bitboard {
    rook_attacks(sq, occupied) | bishop_attacks(sq, occupied)
}

fn slide(sq: Square, occupied: Bitboard, step: fn(Bitboard) -> Bitboard) -> Bitboard {
    let mut result = Bitboard::EMPTY;
    let mut ray = Bitboard::from_square(sq);
    loop {
        ray = step(ray);
        if ray.is_empty() {break}
        result |= ray;
        if !(ray & occupied).is_empty() {break}
    }
    result
}

#[derive(Debug, PartialEq, Eq)]
pub enum FenError {
    WrongFieldCount(usize),
    WrongRankCount(usize),
    InvalidPiece(char),
    RankTooLong(usize),
    RankTooShort(usize),
    InvalidTurn(String),
    InvalidCastling(String),
    InvalidEnPassant(String),
    InvalidNumber(String),
}

impl fmt::Display for FenError {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            FenError::WrongFieldCount(n) => write!(f, "FEN: 6 champs attendus, {n} trouvés"),
            FenError::WrongRankCount(n) => write!(f, "FEN: 8 rangées attendues, {n} trouvées"),
            FenError::InvalidPiece(c) => write!(f, "FEN: pièce inconnue {c:?}"),
            FenError::RankTooLong(r) => write!(f, "FEN: la rangée {r} dépasse 8 cases"),
            FenError::RankTooShort(r) => write!(f, "FEN: la rangée {r} a moins de 8 cases"),
            FenError::InvalidTurn(s) => write!(f, "FEN: trait invalide {s:?}"),
            FenError::InvalidCastling(s) => write!(f, "FEN: roque invalide {s:?}"),
            FenError::InvalidEnPassant(s) => write!(f, "FEN: prise en passant invalide {s:?}"),
            FenError::InvalidNumber(s) => write!(f, "FEN: nombre invalide {s:?}"),
        }
    }
}

impl std::error::Error for FenError {}

impl Board {
    pub const START_FEN: &str = "rnbqkbnr/pppppppp/8/8/8/8/PPPPPPPP/RNBQKBNR w KQkq - 0 1";

    pub fn startpos() -> Board {
        Board::from_fen(Self::START_FEN).expect("FEN de départ valide")
    }

    pub fn from_fen(fen: &str) -> Result<Board,FenError>{
        let fields: Vec<&str> = fen.split_whitespace().collect();
        let &[placement, turn, castling, en_passant, halfmove, fullmove] = fields.as_slice() else {
            return Err(FenError::WrongFieldCount(fields.len()));
        };

        let mut b = Board::empty();
        parse_placement(&mut b, placement)?;
        b.turn = parse_turn(turn)?;
        b.castling = parse_castlings(castling)?;
        b.en_passant = parse_en_passant(en_passant)?;
        b.halfmove = parse_number(halfmove)?;
        b.fullmove = parse_number(fullmove)?;

        Ok(b)
    }
}


fn parse_turn(turn: &str) -> Result<Color,FenError> {
    match turn {
        "w" => Ok(Color::White),
        "b" => Ok(Color::Black),
        _ => Err(FenError::InvalidTurn(turn.to_string()))
    }
}

fn parse_castlings(castling: &str) -> Result<CastlingRights,FenError> {
    let mut castle = CastlingRights::EMPTY;
    
    if castling == "-" {
        return Ok(CastlingRights::EMPTY)
    }

    for c in castling.chars() {
        match c {
            'K' => { castle.white_kingside = true }
            'Q' => { castle.white_queenside = true }
            'k' => { castle.black_kingside = true }
            'q' => { castle.black_queenside = true }
            _ => return Err(FenError::InvalidCastling(castling.to_string()))
        }
    }
    Ok(castle)
}

fn parse_en_passant(text: &str) -> Result<Option<Square>, FenError> {
    if text == "-" {
        return Ok(None);
    }
    match Square::from_uci(text) {
        // rangée "3" = 2 et rangée "6" = 5 en numérotation interne
        Some(sq) if sq.rank() == 2 || sq.rank() == 5 => Ok(Some(sq)),
        _ => Err(FenError::InvalidEnPassant(text.to_string())),
    }
}

fn parse_number(text: &str) -> Result<u16, FenError> {
    text.parse().map_err(|_| FenError::InvalidNumber(text.to_string()))
}

fn parse_placement(board: &mut Board, placement: &str) -> Result<(), FenError> {
    let ranks: Vec<&str> = placement.split('/').collect();
    if ranks.len() != 8 {
        return Err(FenError::WrongRankCount(ranks.len()));
    }

    for (i, rank_text) in ranks.iter().enumerate() {
        let rank = 7 - i; // la 1re rangée du texte est la rangée 8
        let mut file = 0usize;

        for c in rank_text.chars() {
            // un chiffre = autant de cases vides
            if let Some(n) = c.to_digit(10) {
                if n == 0 {
                    return Err(FenError::InvalidPiece(c));
                }
                file += n as usize;
                if file > 8 {
                    return Err(FenError::RankTooLong(rank));
                }
                continue;
            }

            // sinon une pièce : minuscule = noir, majuscule = blanc
            let piece = Piece::from_char(c.to_ascii_lowercase()).ok_or(FenError::InvalidPiece(c))?;
            let color = if c.is_ascii_uppercase() { Color::White } else { Color::Black };

            if file >= 8 {
                return Err(FenError::RankTooLong(rank));
            }
            let Some(sq) = Square::new((rank * 8 + file) as u8) else {
                return Err(FenError::RankTooLong(rank)); // inatteignable : rank et file sont < 8
            };
            board.put_piece(color, piece, sq);
            file += 1;
        }

        if file < 8 {
            return Err(FenError::RankTooShort(rank));
        }
    }
    Ok(())
}

pub struct Board {
    pieces: [[Bitboard; 6]; 2],
    by_color: [Bitboard; 2],
    occupied: Bitboard,

    pub turn: Color,
    pub castling: CastlingRights,
    pub en_passant: Option<Square>,
    pub halfmove: u16,
    pub fullmove: u16,
}

impl Board {
    pub fn empty() -> Board {
        Board {
            pieces: [[Bitboard::EMPTY;6];2],
            by_color: [Bitboard::EMPTY;2],
            occupied: Bitboard::EMPTY,

            turn: Color::White,
            castling: CastlingRights::EMPTY,
            en_passant: None,
            halfmove: 0,
            fullmove: 1
        }
    }

    pub fn pieces(&self, color: Color, piece: Piece) -> Bitboard {
        self.pieces[color.index()][piece.index()]
    }

    pub fn color_bb(&self, color: Color) -> Bitboard {
        self.by_color[color.index()]
    }
    pub fn occupied(&self) -> Bitboard {
        self.occupied
    }

    pub fn put_piece(&mut self, color: Color, piece: Piece, sq: Square) {
        debug_assert!(!self.occupied.contains(sq), "case {sq:?} déjà occupée");
        self.pieces[color.index()][piece.index()].insert(sq);
        self.by_color[color.index()].insert(sq);
        self.occupied.insert(sq);
    }
    pub fn remove_piece(&mut self, sq: Square) -> Option<(Color, Piece)> {
        let (color, piece) = self.piece_at(sq)?;   // case vide : on sort avec None
        self.pieces[color.index()][piece.index()].remove(sq);
        self.by_color[color.index()].remove(sq);
        self.occupied.remove(sq);
        Some((color, piece))
    }

    pub fn piece_at(&self, sq: Square) -> Option<(Color, Piece)> {
        if !self.occupied.contains(sq) {return None}

        let color = if self.by_color[Color::White.index()].contains(sq) {
            Color::White 
        } else {
            Color::Black
        };

        for piece in Piece::ALL {
            if self.pieces[color.index()][piece.index()].contains(sq) {
                return Some((color, piece))
            }
        }
        None
    }
}

#[derive(Clone, Copy, PartialEq, Eq, Hash, Debug)]
pub struct Move {
    pub from: Square,
    pub to: Square,
    pub prom: Option<Piece>
}


impl Move {
    pub fn new(from: Square, to: Square, prom: Option<Piece>) -> Move {
        Move { from, to, prom }
    }

    pub fn from_uci(s: &str) -> Option<Move> {
        if s.len() != 4 && s.len() != 5 {
            return None;
        }

        let from = Square::from_uci(s.get(0..2)?)?;
        let to = Square::from_uci(s.get(2..4)?)?;

        let prom = match s.get(4..)? {
            ""  => None,
            "q" => Some(Piece::Queen),
            "r" => Some(Piece::Rook),
            "b" => Some(Piece::Bishop),
            "n" => Some(Piece::Knight),
            _ => return None,
        };

        Some(Move::new(from, to, prom))
    }

    pub fn to_uci(self) -> String {
        let mut all = String::with_capacity(5);
        all.push_str(&self.from.to_uci());
        all.push_str(&self.to.to_uci());
        if let Some(piece) = self.prom {
            all.push(piece.to_char());
        }
        all
    }
}

use super::Board;
use super::Move;

impl Board {
    pub fn make_move(&mut self, mv: Move) {
        let us = self.turn;
        let old_en_passant = self.en_passant;

        let (_, piece) = self.remove_piece(mv.from).unwrap();
        let captured = self.remove_piece(mv.to);

        // prise en passant : un pion qui change de colonne vers la case de prise en passant
        let is_en_passant = piece == Piece::Pawn
            && Some(mv.to) == old_en_passant
            && mv.from.file() != mv.to.file();
        if is_en_passant {
            let sq = Square::new(mv.from.rank() * 8 + mv.to.file()).unwrap();
            self.remove_piece(sq);
        }

        self.put_piece(us, mv.prom.unwrap_or(piece), mv.to);

        // roque : la tour saute de l'autre côté du roi
        if piece == Piece::King && mv.from.file().abs_diff(mv.to.file()) == 2 {
            let rank = mv.from.rank();
            let (rook_from, rook_to) = if mv.to.file() == 6 { (7, 5) } else { (0, 3) };
            let rook_from = Square::new(rank * 8 + rook_from).unwrap();
            let rook_to = Square::new(rank * 8 + rook_to).unwrap();
            self.remove_piece(rook_from);
            self.put_piece(us, Piece::Rook, rook_to);
        }

        // droits de roque : on regarde départ ET arrivée,
        // ça couvre « la tour part » et « la tour est capturée »
        self.lose_castling_rights(mv.from);
        self.lose_castling_rights(mv.to);

        // double poussée : la case sautée est à mi-chemin
        self.en_passant = if piece == Piece::Pawn && mv.from.rank().abs_diff(mv.to.rank()) == 2 {
            Square::new(((mv.from.index() + mv.to.index()) / 2) as u8)
        } else {
            None
        };

        // règle des 50 coups : remise à zéro sur coup de pion ou capture
        if piece == Piece::Pawn || captured.is_some() || is_en_passant {
            self.halfmove = 0;
        } else {
            self.halfmove += 1;
        }

        if us == Color::Black {
            self.fullmove += 1;
        }
        self.turn = us.opposite();
    }

    fn lose_castling_rights(&mut self, sq: Square) {
        match sq.index() {
            0 => self.castling.white_queenside = false,
            7 => self.castling.white_kingside = false,
            4 => {
                self.castling.white_kingside = false;
                self.castling.white_queenside = false;
            }
            56 => self.castling.black_queenside = false,
            63 => self.castling.black_kingside = false,
            60 => {
                self.castling.black_kingside = false;
                self.castling.black_queenside = false;
            }
            _ => {}
        }
    }

use super::attacks::*;
use super::{Board, Color, Piece, Square};

impl Board {
    pub fn king_square(&self, color: Color) -> Option<Square>{
        self.pieces(color, Piece::King).lsb()
    }

    pub fn is_attacked(&self, sq: Square, by: Color) -> bool {
        let occupied = self.occupied();

        let king = self.pieces(by, Piece::King);
        let knight = self.pieces(by, Piece::Knight);
        let pawn = self.pieces(by, Piece::Pawn);
        let diago = self.pieces(by, Piece::Queen) | self.pieces(by, Piece::Bishop);
        let straight = self.pieces(by, Piece::Queen) | self.pieces(by, Piece::Rook);

        !(king_attacks(sq) & king).is_empty()
            || !(knight_attacks(sq) & knight).is_empty()
            || !(pawn_attacks(by.opposite(), sq) & pawn).is_empty()
            || !(bishop_attacks(sq, occupied) & diago).is_empty()
            || !(rook_attacks(sq, occupied) & straight).is_empty()
    }

    pub fn in_check(&self) -> bool {
        match self.king_square(self.turn) {
            Some(king) => self.is_attacked(king, self.turn.opposite()),
            None => false
        }
    }
}

use super::attacks::*;
use super::{Board, Move, Piece, Color, Bitboard, Square};


impl Board {
    fn king_moves(&self, moves: &mut Vec<Move>) {
        let us = self.turn;
        let own = self.color_bb(us);

        for from in self.pieces(us, Piece::King) {
            let target = king_attacks(from) & !own;
            for to in target {
                moves.push(Move::new(from, to, None));
            }
        }
    }

    fn queen_moves(&self, moves: &mut Vec<Move>) {
        let us = self.turn;
        let own = self.color_bb(us);

        for from in self.pieces(us, Piece::Queen) {
            let target = queen_attacks(from, self.occupied()) & !own;
            for to in target {
                moves.push(Move::new(from, to, None));
            }
        }
    }

    fn rook_moves(&self, moves: &mut Vec<Move>) {
        let us = self.turn;
        let own = self.color_bb(us);

        for from in self.pieces(us, Piece::Rook) {
            let target = rook_attacks(from, self.occupied()) & !own;
            for to in target {
                moves.push(Move::new(from, to, None));
            }
        }
    }

    fn bishop_moves(&self, moves: &mut Vec<Move>) {
        let us = self.turn;
        let own = self.color_bb(us);

        for from in self.pieces(us, Piece::Bishop) {
            let target = bishop_attacks(from, self.occupied()) & !own;
            for to in target {
                moves.push(Move::new(from, to, None));
            }
        }
    }

    fn knight_moves(&self, moves: &mut Vec<Move>) {
        let us = self.turn;
        let own = self.color_bb(us);

        for from in self.pieces(us, Piece::Knight) {
            let target = knight_attacks(from) & !own;
            for to in target {
                moves.push(Move::new(from, to, None));
            }
        }
    }

    fn pawn_moves(&self, moves: &mut Vec<Move>) {
        let us = self.turn;
        let empty = !self.occupied();

        let (start_rank, promo_rank) = match us {
            Color::White => (1,7),
            Color::Black => (6,0)
        };

        let forward = |b: Bitboard| match us {
            Color::White => b.north(),
            Color::Black => b.south(),
        };

        let mut target = self.color_bb(us.opposite());
        if let Some(ep) = self.en_passant { target.insert(ep) }

        for from in self.pieces(us, Piece::Pawn) {
            let one = forward(Bitboard::from_square(from)) & empty;

            for to in one {
                push_pawn_move(moves, from, to, promo_rank);
            }

            if from.rank() == start_rank {
                let two = forward(one) & empty;
                for to in two {
                    moves.push(Move::new(from, to, None));
                }
            }

            for to in target & pawn_attacks(us, from) {
                push_pawn_move(moves, from, to, promo_rank);
            } 
        }
    }

    fn castle_moves(&self, moves: &mut Vec<Move>) {
        let us = self.turn;
        let them = us.opposite();

        // (droit petit roque, droit grand roque, index de la première case de la rangée de base)
        let (kingside, queenside, base) = match us {
            Color::White => (self.castling.white_kingside, self.castling.white_queenside, 0u8),
            Color::Black => (self.castling.black_kingside, self.castling.black_queenside, 56u8),
        };
        let sq = |file: u8| Square::new(base + file).unwrap(); // la case de cette colonne sur la rangée de base
        let occupied = self.occupied();

        // le roi doit être sur sa case (colonne 4) et ne pas être en échec
        if !self.pieces(us, Piece::King).contains(sq(4)) || self.is_attacked(sq(4), them) {
            return;
        }

        // petit roque : tour en colonne 7, cases f et g (5 et 6) vides, f non attaquée
        if kingside
            && self.pieces(us, Piece::Rook).contains(sq(7))
            && !occupied.contains(sq(5))
            && !occupied.contains(sq(6))
            && !self.is_attacked(sq(5), them)
        {
            moves.push(Move::new(sq(4), sq(6), None));
        }

        if queenside
            && self.pieces(us, Piece::Rook).contains(sq(0))
            && !occupied.contains(sq(1))
            && !occupied.contains(sq(2))
            && !occupied.contains(sq(3))
            && !self.is_attacked(sq(3), them) 
        {
            moves.push(Move::new(sq(4), sq(2), None));
        }
    }

    pub fn pseudo_legal_moves(&self) -> Vec<Move> {
        let mut moves = Vec::with_capacity(64);
        self.pawn_moves(&mut moves);
        self.knight_moves(&mut moves);
        self.bishop_moves(&mut moves);
        self.rook_moves(&mut moves);
        self.queen_moves(&mut moves);
        self.king_moves(&mut moves);
        self.castle_moves(&mut moves);
        moves
    }

    pub fn legal_moves(&self) -> Vec<Move> {
        let us = self.turn; // à lire AVANT make_move : le trait change après
        let mut legal = Vec::with_capacity(48);

        for mv in self.pseudo_legal_moves() {
            let mut next = self.clone();
            next.make_move(mv);

            let safe = match next.king_square(us) {
                Some(king) => !next.is_attacked(king, us.opposite()),
                None => true,
            };
            if safe {
                legal.push(mv);
            }
        }
        legal
    }
}



fn push_pawn_move(moves: &mut Vec<Move>, from: Square, to: Square, promo_rank: u8) {
    if to.rank() == promo_rank {
        for piece in [Piece::Queen, Piece::Rook, Piece::Bishop, Piece::Knight] {
            moves.push(Move::new(from, to, Some(piece)));
        }
    } else {
        moves.push(Move::new(from, to, None));
    }
}

use super::Board;

impl Board {
    pub fn _perft(&self, depth: u32) -> u64 {
        if depth == 0 {
            return 1;
        }
        let moves = self.legal_moves();
        if depth == 1 {
            return moves.len() as u64; // inutile de jouer le dernier coup
        }
        moves
            .iter()
            .map(|&mv| {
                let mut next = self.clone();
                next.make_move(mv);
                next._perft(depth - 1)
            })
            .sum()
    }

    /// Le perft de chaque coup de la racine : sert à localiser un bug.
    pub fn _divide(&self, depth: u32) -> Vec<(String, u64)> {
        let mut result: Vec<(String, u64)> = self
            .legal_moves()
            .iter()
            .map(|&mv| {
                let mut next = self.clone();
                next.make_move(mv);
                (mv.to_uci(), next._perft(depth - 1))
            })
            .collect();
        result.sort();
        result
    }
}

use std::error::Error;
use std::sync::Arc;

use futures_util::StreamExt;

use litchee::LichessClient;
use litchee::api::gameplay::board::{LichessBoardEvent, LichessGameEventInfo, LichessGameState, LichessIncomingEvent};
use litchee::api::gameplay::challenges::LichessChallenge;
use litchee::model::LichessColor;

use crate::bots::{BotTrait, BotRandom, };//BotMiniMax};
use crate::engine::{Move, Color, board::*};


pub struct Api {
    client: Arc<LichessClient>
}

impl Api {
    pub fn new() -> Self {
        let token = std::env::var("LICHESS_TOKEN").expect("Pas de token .env");
        let client = LichessClient::builder().token(token).build().expect("Token error");
        Api { client: Arc::new(client) }
    }

    pub async fn loop_event(&self) -> Result<(), Box<dyn Error>>{
        let mut stream = self.client.bot().stream_events().await?;

        while let Some(event) = stream.next().await {
            match event? {
                LichessIncomingEvent::Challenge { challenge } => {
                    if let Err(e) = self.handle_challenge(challenge).await {
                        eprintln!("Défi ignoré : {e}");
                    }
                }
                LichessIncomingEvent::GameStart { game } => {
                    if let Err(e) = self.handle_start(game).await {
                        eprintln!("Démarrage de partie échoué : {e}");
                    }
                }
                LichessIncomingEvent::GameFinish { game } => {
                    let _game = game;
                }
                LichessIncomingEvent::ChallengeCanceled { challenge } => {
                    let _challenge = challenge;
                }
                LichessIncomingEvent::ChallengeDeclined { challenge } => {
                    let _challenge = challenge;
                }
                _ => {}
            }
        }
        Ok(())
    }

    async fn handle_challenge(&self, chal: Box<LichessChallenge>) -> Result<(), Box<dyn Error>> {
        self.client.challenges().accept(&chal.id, None).await?;
        Ok(())
    }

    async fn handle_start(&self, game_info: LichessGameEventInfo) -> Result<(), Box<dyn Error>> {
        let color = match game_info.color.unwrap_or(LichessColor::White) {
            LichessColor::White => Color::White,
            LichessColor::Black => Color::Black
        };

        let client_clone = Arc::clone(&self.client);
        let Some(id) = game_info.id else {
            eprintln!("GameStart sans identifiant");
            return Ok(());
        };

        tokio::spawn(async move {
            let mut game = Game::new(client_clone, id, color, BotRandom);

            game.main_loop().await.unwrap_or_else(|e| {
                eprintln!("Erreur critique dans la partie {}: {}", game.id, e);
            });
        });
        Ok(())
    }
}


#[derive(Debug)]
struct Game<B: BotTrait> {
    client: Arc<LichessClient>,
    id: String,
    board: Board,
    color: Color,
    board_move: usize,
    sent_move: Option<usize>,
    bot: B
}

impl <B: BotTrait> Game<B>{
    fn new(client: Arc<LichessClient>, id: String, color: Color, bot: B) -> Self {
        Game {
            client,
            id,
            board: Board::startpos(),
            color,
            board_move: 0,
            sent_move: None,
            bot
        }
    }

    async fn main_loop(&mut self) -> Result<(), Box<dyn Error>> {
        let mut game_stream = self.client.bot().stream_game(&self.id).await?;

        while let Some(game_event) = game_stream.next().await {
            match game_event? {
                LichessBoardEvent::GameFull(game_full) => {
                    self.check_state(game_full.state).await?;
                }
                LichessBoardEvent::GameState(state) => {
                    self.check_state(state).await?;
                }
                LichessBoardEvent::OpponentGone(gone) => {
                    println!("{:#?}", gone);
                }
                LichessBoardEvent::ChatLine(chat) => {
                    println!("{:#?}", chat);
                }
                _ => {}
            }
        }
        Ok(())
    } 

    async fn check_state(&mut self, state: LichessGameState) -> Result<(), Box<dyn Error>>{
        let movs: Vec<&str> = state.moves.split_whitespace().collect();

        for m in &movs[self.board_move..] {
            let mov_to_play = Move::from_uci(m).ok_or_else(|| format!("coup illisible : {m}"))?;
            self.board.make_move(mov_to_play);
        }

        self.board_move = movs.len();

        if self.color == self.board.turn && self.sent_move != Some(self.board_move) {
            if let Some(mv) = self.bot.play_move(&self.board) {
                match self.play_mov(mv).await {
                    Ok(()) => self.sent_move = Some(self.board_move),
                    Err(e) => eprintln!("Partie {} : coup refusé : {e}", self.id),
                }
            }
        }
        Ok(())
    }

    async fn play_mov(&self,mov: Move) -> Result<(), Box<dyn Error>> {
        let mov = mov.to_uci();

        self.client.bot().make_move(&self.id, &mov, false).await?;
        Ok(())
    }    
}
